cmake_minimum_required(VERSION 3.15)
project(TrajectoryODESolver LANGUAGES Fortran)

set(CMAKE_Fortran_STANDARD 2008)
set(CMAKE_Fortran_STANDARD_REQUIRED ON)

set(SOURCES
    vector_m.f90
    diff_m.f90
    forces_m.f90
    int_m.f90
    main.f90
)

add_executable(trajectory_solver ${SOURCES})

if (CMAKE_Fortran_COMPILER_ID MATCHES "GNU")
    target_compile_options(trajectory_solver PRIVATE -O3 -Wall -Wextra)
elseif (CMAKE_Fortran_COMPILER_ID MATCHES "Intel")
    target_compile_options(trajectory_solver PRIVATE /O3)
endif()
